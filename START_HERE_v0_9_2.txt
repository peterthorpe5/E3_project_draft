INSTALL AND PROVE THE ACTIVE SOURCE
-----------------------------------

From the unpacked repository:

    cd E3_project_draft/e3_end_to_end_workflow

    conda run --name e3_end_to_end_workflow \
        python -m pip install \
        --no-deps \
        --force-reinstall \
        --editable "$(pwd)"

    conda run --name e3_end_to_end_workflow \
        e3-workflow diagnose-install \
        --source-root "$(pwd)" \
        --require-source-match

The diagnostic must report MATCHED_SOURCE and version 0.9.2.
